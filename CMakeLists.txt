cmake_minimum_required(VERSION 3.20)
project(helpdesk_pro LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(httplib CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(OpenSSL REQUIRED)
find_path(MYSQL_INCLUDE_DIR NAMES mysql.h PATH_SUFFIXES mysql mariadb REQUIRED)
find_library(MYSQL_LIBRARY NAMES libmysql mysqlclient libmariadb mariadb REQUIRED)
add_executable(helpdesk src/main.cpp)
target_include_directories(helpdesk PRIVATE ${MYSQL_INCLUDE_DIR})
target_link_libraries(helpdesk PRIVATE httplib::httplib nlohmann_json::nlohmann_json OpenSSL::Crypto ${MYSQL_LIBRARY})
enable_testing()
add_executable(validation_test tests/validation.cpp)
target_include_directories(validation_test PRIVATE src)
target_link_libraries(validation_test PRIVATE nlohmann_json::nlohmann_json)
add_test(NAME validation COMMAND validation_test)
